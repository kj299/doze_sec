# benign_corpus.txt -- known-BENIGN artifacts that look like attacks.
#
# Read by tools/benign_corpus_check.ps1. Every entry is a false-positive
# twin: something a real machine carries that resembles an indicator, and
# that the audit must report as context (OK/INFO), or as a finding a person
# can adjudicate with the note below. Two of these fired as ACTIVE COMPROMISE
# on the owner's own machine in 2026-08 (P9NP, Event 104) because they had
# never been tested; a third (never-onboarded Sense) contradicted its own
# summary. For this audience a false "you are compromised" is the most
# harmful output the tool can produce.
#
# FORMAT -- INI-style blocks (regex-safe; the '|' corpora cannot hold regexes):
#   [id]                    short slug, unique
#   class     = QUIET       the tool prints a CONTEXT line for the benign twin;
#                           every report line matching `signature` must carry
#                           a leading tag <= maxsev (OK < INFO < WARNING < CRITICAL).
#                           -Mode Report FAILS on a higher tag.
#             = ADVISE      `signature` is the FINDING-shaped line that has a
#                           known benign cause; on a real machine a match prints
#                           the note (how to verify). The human adjudicates;
#                           never a failure.
#   maxsev    = OK|INFO|WARNING   (QUIET only)
#   proof     = harness:<Name substring of an Invert=$true case in tests/detection_selftest.ps1>
#             | ci:<step-name substring in .github/workflows/windows-smoke.yml>
#             | field:<why this cannot be planted -- observable only on a real machine>
#             several proofs: separate with ';'
#   signature = .NET regex matched per report line
#   note      = origin, and how a person verifies it on their machine
#
# -Mode Lint fails on: an entry without class/proof/signature/note, an
# unresolvable proof, an invalid regex, a harness Invert case that no entry
# catalogues (every false-positive guard must be documented here), or fewer
# than 10 entries (vacuous). -Mode Report is run by tests/field_test.ps1 on a
# real machine and by the harness on its own report.

# ---- QUIET: the audit prints context, and must keep it at or below maxsev ----

[netprov-p9np]
class     = QUIET
maxsev    = INFO
proof     = ci:Network provider: Microsoft-signed System32 provider is context
signature = Non-default network provider .* Microsoft-signed system DLL
note      = WSL ships p9np.dll and registers P9NP as a network provider on every WSL machine. A Microsoft-signed provider under System32 is context, not credential capture. Field false positive 2026-08-30, fixed in #184. Verify: (Get-ItemProperty 'HKLM:\SYSTEM\CurrentControlSet\Control\NetworkProvider\Order').ProviderOrder and Get-AuthenticodeSignature on the ProviderPath.

[sense-never-onboarded]
class     = QUIET
maxsev    = INFO
proof     = ci:edr_presence.ps1
signature = never been onboarded to Defender for Endpoint
note      = Windows ships the Defender for Endpoint sensor service ('Sense') inert on every machine never onboarded to MDE; a stopped Sense on a home PC is how Windows ships. Field false positive 2026-08-29, fixed in #183. Verify: HKLM:\SOFTWARE\Microsoft\Windows Advanced Threat Protection\Status OnboardingState (absent or 0 = never onboarded).

# ---- ADVISE: a finding with a known benign cause; the person adjudicates ----

[ev104-smartcard]
class     = ADVISE
proof     = ci:Event 104 provider filter
signature = System event log was cleared
note      = Event IDs are unique only per provider. Smart-card readers (WudfUsbccidDriver) log Event 104 too; the check filters to Microsoft-Windows-Eventlog since #184. If this still fires, verify: Get-WinEvent -FilterHashtable @{LogName='System';Id=104;ProviderName='Microsoft-Windows-Eventlog'} -MaxEvents 5 -- a genuine clear shows here, and the log-gap check nearby would show a record-number break.

[startup-signed-lnk]
class     = ADVISE
proof     = harness:Benign signed shortcut in Startup
signature = Startup item .*\.lnk
note      = OneDrive, Teams and vendor updaters drop .lnk files into Startup constantly; a shortcut to a Microsoft-signed binary is not flagged. Verify the target: Get-AuthenticodeSignature on what the shortcut points at.
